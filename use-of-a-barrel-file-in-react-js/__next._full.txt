1d:T1159,<h2>Use of a barrel file (<code>index.js</code> or <code>index.ts</code>) with in modern react codebases.</h2>
<blockquote>
<p><strong>A barrel file is nothing just a <code>index.js</code> or <code>index.ts</code> file within a folder.</strong></p>
</blockquote>
<p>It is useful to make <code>import</code> statements shorter and beautiful.</p>
<pre><code class="hljs language-js"><span class="hljs-comment">// src/components/avatar/index.js</span>

<span class="hljs-keyword">export</span> * <span class="hljs-keyword">from</span> <span class="hljs-string">"./avatar.jsx"</span>;

<span class="hljs-keyword">export</span> { classes <span class="hljs-keyword">as</span> avatarClasses } <span class="hljs-keyword">from</span> <span class="hljs-string">"./classes.js"</span>;
</code></pre>
<pre><code class="hljs language-js"><span class="hljs-comment">// src/components/avatar/avatar.jsx</span>

<span class="hljs-keyword">export</span> <span class="hljs-keyword">function</span> <span class="hljs-title function_">Avatar</span>(<span class="hljs-params"></span>) {
  <span class="hljs-keyword">return</span> (
    <span class="hljs-comment">// Your component</span>
  )
}

<span class="hljs-comment">// src/components/avatar/classes.js</span>

<span class="hljs-keyword">export</span> <span class="hljs-keyword">const</span> classes = {
  <span class="hljs-attr">root</span>: <span class="hljs-string">'avatar__root'</span>,
}
</code></pre>
<pre><code class="hljs language-jsx"><span class="hljs-comment">// with barrel file</span>
<span class="hljs-keyword">import</span> { <span class="hljs-title class_">Avatar</span> } <span class="hljs-keyword">from</span> <span class="hljs-string">"@components/avatar"</span>;
